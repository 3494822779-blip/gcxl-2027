cmake_minimum_required(VERSION 3.16)
project(block_deepstream)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(block_interfaces REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0 gstreamer-app-1.0)
set(DEEPSTREAM_ROOT /opt/nvidia/deepstream/deepstream CACHE PATH "DeepStream SDK")
add_executable(perception src/nvmm_perception.cpp)
target_include_directories(perception PRIVATE ${DEEPSTREAM_ROOT}/sources/includes /usr/local/cuda/include)
target_link_directories(perception PRIVATE ${DEEPSTREAM_ROOT}/lib)
target_link_libraries(perception PkgConfig::GST nvds_meta nvdsgst_meta)
ament_target_dependencies(perception rclcpp rcl_interfaces block_interfaces)
set_target_properties(perception PROPERTIES INSTALL_RPATH ${DEEPSTREAM_ROOT}/lib)
target_compile_options(perception PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS perception DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_target_cache test/test_target_cache.cpp)
  target_include_directories(test_target_cache PRIVATE src)
  ament_target_dependencies(test_target_cache block_interfaces)
  target_link_libraries(test_target_cache pthread)
  add_test(NAME target_cache COMMAND test_target_cache)
endif()
ament_package()
