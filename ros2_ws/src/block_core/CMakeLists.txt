cmake_minimum_required(VERSION 3.16)
project(block_core)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake QUIET)
find_package(jsoncpp REQUIRED)
include(CTest)
add_library(${PROJECT_NAME} src/core.cpp src/task_memory.cpp src/search.cpp src/sweep_control.cpp src/claw_opening.cpp)
target_link_libraries(${PROJECT_NAME} PUBLIC jsoncpp_lib)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> )
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
install(DIRECTORY include/ DESTINATION include)
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  if(ament_cmake_FOUND)
    find_package(ament_cmake_gtest REQUIRED)
    ament_add_gtest(test_core test/test_core.cpp test/test_search.cpp test/test_sweep_control.cpp test/test_claw_opening.cpp)
  else()
    find_package(GTest REQUIRED)
    add_executable(test_core test/test_core.cpp test/test_search.cpp test/test_sweep_control.cpp test/test_claw_opening.cpp)
    target_link_libraries(test_core GTest::gtest_main)
    add_test(NAME test_core COMMAND test_core)
  endif()
  target_link_libraries(test_core ${PROJECT_NAME} util)
endif()
if(ament_cmake_FOUND)
  ament_export_dependencies(jsoncpp)
  ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
  ament_export_include_directories(include)
  ament_package()
endif()
