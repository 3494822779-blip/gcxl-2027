cmake_minimum_required(VERSION 3.16)
project(serial_minimal)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(block_core REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(serial_node src/serial_node.cpp)
ament_target_dependencies(serial_node block_core rclcpp std_msgs)
install(TARGETS serial_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_serial_bridge test/test_serial_bridge.cpp)
  ament_target_dependencies(test_serial_bridge rclcpp std_msgs block_core)
  target_link_libraries(test_serial_bridge util)
endif()
ament_package()
