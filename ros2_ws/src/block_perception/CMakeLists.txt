cmake_minimum_required(VERSION 3.16)
project(block_perception)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(block_interfaces REQUIRED)
find_package(block_core REQUIRED)
include_directories(include)
add_library(perception_core INTERFACE)
target_include_directories(perception_core INTERFACE include)
ament_target_dependencies(perception_core INTERFACE rclcpp block_interfaces block_core)
add_executable(selector src/selector_node.cpp)
add_executable(filter src/filter_node.cpp)
add_executable(alignment src/alignment_node.cpp)
add_executable(gripper src/gripper_node.cpp)
foreach(node selector filter alignment gripper)
  target_compile_definitions(${node} PRIVATE BLOCK_NODE_NAME="${node}")
  ament_target_dependencies(${node} rclcpp rclcpp_action sensor_msgs std_msgs block_interfaces block_core)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_nodes test/test_nodes.cpp)
  ament_target_dependencies(test_nodes rclcpp sensor_msgs std_msgs block_interfaces block_core)
endif()

ament_package()
