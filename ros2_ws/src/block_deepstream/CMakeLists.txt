cmake_minimum_required(VERSION 3.16)
project(block_deepstream)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(block_interfaces REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0 gstreamer-app-1.0)
set(DEEPSTREAM_ROOT /opt/nvidia/deepstream/deepstream CACHE PATH "DeepStream SDK")
foreach(node detector tracker)
  add_executable(${node} src/bridge.cpp)
  if(node STREQUAL "tracker")
    target_compile_definitions(${node} PRIVATE TRACKER_NODE=1)
  endif()
  target_include_directories(${node} PRIVATE ${DEEPSTREAM_ROOT}/sources/includes /usr/local/cuda/include)
  target_link_directories(${node} PRIVATE ${DEEPSTREAM_ROOT}/lib)
  target_link_libraries(${node} PkgConfig::GST nvds_meta nvdsgst_meta)
  ament_target_dependencies(${node} rclcpp rcl_interfaces sensor_msgs block_interfaces)
  set_target_properties(${node} PROPERTIES INSTALL_RPATH ${DEEPSTREAM_ROOT}/lib)
  target_compile_options(${node} PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
ament_package()
