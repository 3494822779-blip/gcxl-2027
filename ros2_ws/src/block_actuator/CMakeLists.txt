cmake_minimum_required(VERSION 3.16)
project(block_actuator)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(block_interfaces REQUIRED)
find_package(block_core REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(sensor_msgs REQUIRED)
include_directories(include)
foreach(node actuator task_executor)
  add_executable(${node} src/${node}_node.cpp)
  target_compile_definitions(${node} PRIVATE BLOCK_NODE_NAME="${node}")
  ament_target_dependencies(${node} rclcpp rclcpp_action block_interfaces block_core std_msgs std_srvs sensor_msgs)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_actions test/test_actions.cpp)
  ament_target_dependencies(test_actions rclcpp rclcpp_action block_interfaces block_core std_msgs std_srvs sensor_msgs)
endif()

ament_package()
